<objective>
Add UUID-aware service functions for folder writes and fix the broken dual-source read path.

Purpose: The folder assignment service layer currently only writes to `folder_assignments` (a table with a FK to `fathom_raw_calls`). Non-Fathom recordings (Zoom, manual paste, MCP import) have no row in `fathom_raw_calls`, so writing their UUID there violates the FK — but `parseInt(uuid)` produces `NaN`, which is silently filtered, so no write ever happens and a false success toast fires. This plan adds the canonical UUID write path and fixes the dual-source read to match the already-correct `getWorkspaceFolderRecordingIds` pattern.

Output: Two new exported functions in `folders.service.ts`, fixed `getRecordingIdsForFolderFilter` and new `getAssignedWorkspaceEntryFolderUuids` in `transcript-filters.service.ts`.
</objective>

```
export async function getRecordingIdsForFolderFilter(folderIds: string[]): Promise<string[]> {
  const allFolderIds = await getFolderAndChildIds(folderIds)
  if (allFolderIds.length === 0) return []

  // Source 1: workspace_entries.folder_id (canonical, UUID-keyed)
  const { data: wsEntries, error: wsError } = await supabase
    .from('workspace_entries')
    .select('recording_id')
    .in('folder_id', allFolderIds)
  if (wsError) throw wsError

  // Source 2: folder_assignments (legacy, BIGINT-keyed)
  const { data: legacyAssigns, error: legacyError } = await supabase
    .from('folder_assignments')
    .select('call_recording_id')
    .in('folder_id', allFolderIds)
  if (legacyError) throw legacyError

  const legacyIds = (legacyAssigns || []).map(a => a.call_recording_id)
  const legacyUuids = await resolveLegacyRecordingIdsToUuids(legacyIds)

  return Array.from(new Set([
    ...(wsEntries || []).map(e => e.recording_id),
    ...legacyUuids,
  ]))
}
```

```
export async function getAssignedWorkspaceEntryFolderUuids(): Promise<Set<string>> {
  const { data: wsEntries, error: wsError } = await supabase
    .from('workspace_entries')
    .select('recording_id')
    .not('folder_id', 'is', null)
  if (wsError) throw wsError

  const { data: legacyAssigns, error: legacyError } = await supabase
    .from('folder_assignments')
    .select('call_recording_id')
  if (legacyError) throw legacyError

  const legacyIds = (legacyAssigns || []).map(a => a.call_recording_id)
  const legacyUuids = await resolveLegacyRecordingIdsToUuids(legacyIds)

  return new Set([
    ...(wsEntries || []).map(e => e.recording_id),
    ...legacyUuids,
  ])
}
```
