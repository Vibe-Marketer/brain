```typescript
import { describe, it, expect, beforeAll, afterAll } from 'vitest';
import { createClient, type SupabaseClient } from '@supabase/supabase-js';
import { randomUUID } from 'crypto';
import { performance } from 'perf_hooks';

const SUPABASE_URL = process.env.SUPABASE_URL || process.env.VITE_SUPABASE_URL;
const SUPABASE_SERVICE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY;
const integrationDbReachable = Boolean(SUPABASE_URL && SUPABASE_SERVICE_KEY);

const SEED_COUNT = 5000;
const QUERY_COUNT = 100;
const P95_TARGET_MS = 200;
const SEED_PREFIX = `phase39bench-${randomUUID().slice(0, 8)}`;

describe.skipIf(!integrationDbReachable)('Phase 39 — useGlobalSearch p95 benchmark', () => {
  let supabase: SupabaseClient;
  const testUserId = randomUUID();
  const testOrgId = randomUUID();
  const seedTitles = ['call about ' + SEED_PREFIX, 'meeting with team', 'standup notes', 'demo recap', 'retro discussion'];
  const seedKeywords = ['authentication', 'pricing', 'integration', 'roadmap', 'onboarding', SEED_PREFIX];

  beforeAll(async () => {
    supabase = createClient(SUPABASE_URL!, SUPABASE_SERVICE_KEY!, { auth: { autoRefreshToken: false, persistSession: false } });

    // Create test org + user
    const { error: orgErr } = await supabase
      .from('organizations')
      .insert({ id: testOrgId, name: 'phase39-bench-org', owner_user_id: testUserId } as never);
    // Insert may fail if RLS / schema mismatch — log but continue.
    if (orgErr) console.warn('[p95] org seed:', orgErr.message);

    // Seed 5000 recordings rows
    const batch: Array<Record<string, unknown>> = [];
    for (let i = 0; i < SEED_COUNT; i++) {
      const includeBenchKey = i % 10 === 0;
      const kw = seedKeywords[i % seedKeywords.length];
      const title = `${seedTitles[i % seedTitles.length]} #${i}${includeBenchKey ? ' ' + SEED_PREFIX : ''}`;
      const transcript = `[00:00:00] Speaker A: This is a meeting about ${kw}. ` +
        (includeBenchKey ? `Note: ${SEED_PREFIX} mentioned. ` : '') +
        `[00:01:00] Speaker B: We discussed ${kw} extensively. `.repeat(20);
      batch.push({
        organization_id: testOrgId,
        owner_user_id: testUserId,
        title,
        full_transcript: transcript,
        summary: `Summary of ${kw} call #${i}`,
        source_app: 'fathom',
        source_call_id: `bench-${i}-${SEED_PREFIX}`,
        recording_start_time: new Date(Date.now() - i * 60_000).toISOString(),
        duration: 1800,
        created_at: new Date(Date.now() - i * 60_000).toISOString(),
        legacy_recording_id: 8_000_000_000 + i,
      });
    }
    // Insert in chunks of 500 to avoid request size limits
    for (let i = 0; i < batch.length; i += 500) {
      const slice = batch.slice(i, i + 500);
      const { error } = await supabase.from('recordings').insert(slice as never);
      if (error) {
        console.warn(`[p95] seed batch ${i}: ${error.message}`);
        break;
      }
    }
  }, 120_000); // 120s timeout for seed

  afterAll(async () => {
    if (!supabase) return;
    await supabase.from('recordings').delete().like('source_call_id', `bench-%-${SEED_PREFIX}`);
    await supabase.from('organizations').delete().eq('id', testOrgId);
  }, 60_000);

  it(`p95 of ${QUERY_COUNT} searches over ${SEED_COUNT}-row mirror is < ${P95_TARGET_MS}ms`, async () => {
    const queries = ['authentication', 'pricing', 'integration', 'roadmap', 'onboarding', SEED_PREFIX, 'standup', 'demo', 'retro', 'call'];
    const latenciesMs: number[] = [];

    for (let i = 0; i < QUERY_COUNT; i++) {
      const term = queries[i % queries.length];
      const escaped = term.replace(/[%_]/g, (m) => `\\${m}`);
      const start = performance.now();
      const { error } = await supabase
        .from('recordings')
        .select('id, legacy_recording_id, title, full_transcript, summary, source_app, source_metadata, duration, recording_start_time, created_at')
        .or(`title.ilike.%${escaped}%,full_transcript.ilike.%${escaped}%,summary.ilike.%${escaped}%`)
        .eq('organization_id', testOrgId)
        .order('created_at', { ascending: false })
        .limit(20);
      const end = performance.now();
      if (error) {
        console.warn(`[p95] query ${i} (${term}):`, error.message);
        continue;
      }
      latenciesMs.push(end - start);
    }

    expect(latenciesMs.length).toBeGreaterThan(QUERY_COUNT * 0.9); // tolerate up to 10% failures
    latenciesMs.sort((a, b) => a - b);
    const p50 = latenciesMs[Math.floor(latenciesMs.length * 0.5)];
    const p95 = latenciesMs[Math.floor(latenciesMs.length * 0.95)];
    const p99 = latenciesMs[Math.floor(latenciesMs.length * 0.99)];

    console.log(`[p95 benchmark] n=${latenciesMs.length} p50=${p50.toFixed(1)}ms p95=${p95.toFixed(1)}ms p99=${p99.toFixed(1)}ms`);

    expect(p95).toBeLessThan(P95_TARGET_MS);
  }, 180_000); // 180s timeout for the 100-query loop
});
```
