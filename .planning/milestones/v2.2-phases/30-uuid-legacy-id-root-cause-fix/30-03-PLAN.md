`src/test/integration-setup.ts`:
```ts
import { createClient, SupabaseClient } from '@supabase/supabase-js';

const TEST_URL = process.env.VITE_SUPABASE_TEST_URL || process.env.VITE_SUPABASE_URL;
const TEST_SERVICE_KEY = process.env.SUPABASE_TEST_SERVICE_ROLE_KEY;

export const integrationDbReachable = Boolean(TEST_URL && TEST_SERVICE_KEY);

export function makeIntegrationClient(): SupabaseClient {
  if (!integrationDbReachable) {
    throw new Error('Integration DB not configured — set SUPABASE_TEST_SERVICE_ROLE_KEY');
  }
  return createClient(TEST_URL!, TEST_SERVICE_KEY!, {
    auth: { autoRefreshToken: false, persistSession: false },
  });
}
```

2. **Test cases:**
   - `assignCallToFolder(999999999, folderId, userId, workspaceId)` succeeds.

   - Delete the folder_assignments rows, workspace_entries rows, recordings row, fathom_raw_calls row, folder, workspace, org (in dependency order).
