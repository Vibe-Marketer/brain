<task id="37-01-T02">
<action>
Run an automated audit script that produces the per-function table. Use this exact bash, saved as `/tmp/audit-edge-functions.sh`, then redirect its output into the audit doc:

```bash
#!/bin/bash
cd /Users/Naegele/dev/brain
for d in supabase/functions/*/; do
  name=$(basename "$d")
  [[ "$name" == "_shared" ]] && continue
  f="$d/index.ts"
  [[ -f "$f" ]] || continue
  has_options=$(grep -c "req.method === 'OPTIONS'" "$f")
  has_jwt=$(grep -cE "authenticateRequest|authHeader\.replace|getUser\(token\)" "$f")
  has_zod=$(grep -c "from 'https://esm.sh/zod" "$f")
  has_org_check=$(grep -cE "organization_memberships|org_id|user_id" "$f")
  has_svc_role=$(grep -c "SUPABASE_SERVICE_ROLE_KEY" "$f")
  has_rationale=$(grep -c "service-role required:" "$f")
  has_idempotency=$(grep -c "processed_webhooks\|already_processed" "$f")
  has_hmac=$(grep -cE "timingSafeEqual|validateEvent" "$f")
  has_replay=$(grep -cE "MAX_AGE_MS|x-zm-request-timestamp|svix" "$f")
  has_html_escape=$(grep -c "escapeHtml" "$f")
  has_magic_bytes=$(grep -c "validateMagicBytes" "$f")
  echo "| $name | $has_options | $has_jwt | $has_zod | $has_org_check | $has_svc_role | $has_rationale | $has_idempotency | $has_hmac | $has_replay | $has_html_escape | $has_magic_bytes |"
done
```
