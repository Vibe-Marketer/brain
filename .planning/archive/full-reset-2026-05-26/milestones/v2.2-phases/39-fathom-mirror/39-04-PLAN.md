# Plan 39-04: Daily Reconcile Cron via pg_cron

## Objective

Schedule `fathom-reconcile` to fire daily at 07:00 UTC via Supabase's `pg_cron` extension calling `net.http_post()`. The schedule passes an `X-Reconcile-Secret` header that the function validates against `RECONCILE_SECRET` env var.

Pattern verified from existing `20260110000008_scheduled_rules_fields.sql:280-298` (automation-scheduler cron).

The migration must be idempotent: unschedule first if existing, then schedule. Gracefully no-op if `pg_cron` is unavailable (documented fallback to Supabase Scheduled Functions dashboard per CONTEXT.md coordination caveat).

```sql
-- Migration: Fathom daily reconcile cron (Phase 39)
-- Purpose:   Schedule fathom-reconcile edge function via pg_cron at 07:00 UTC daily.
-- Date:      2026-05-12
-- Author:    Claude (Phase 39 plan 39-04)
--
-- Idempotent: unschedules existing job first, then schedules new one.
-- Graceful: if pg_cron extension unavailable (Free tier), emits NOTICE and continues.
-- Fallback: operator must schedule via Supabase Scheduled Functions dashboard.

-- ============================================================================
-- ENABLE pg_cron + pg_net extensions (idempotent)
-- ============================================================================
CREATE EXTENSION IF NOT EXISTS pg_net WITH SCHEMA extensions;
CREATE EXTENSION IF NOT EXISTS pg_cron WITH SCHEMA extensions;

-- ============================================================================
-- UNSCHEDULE existing job (idempotent - safe to re-run)
-- ============================================================================
DO $outer$
BEGIN
  PERFORM cron.unschedule('fathom-daily-reconcile');
  RAISE NOTICE 'Unscheduled existing fathom-daily-reconcile job';
EXCEPTION
  WHEN undefined_function THEN
    RAISE NOTICE 'pg_cron not available - skipping unschedule';
  WHEN OTHERS THEN
    NULL;
END $outer$;

-- ============================================================================
-- SCHEDULE new daily reconcile job at 07:00 UTC
-- ============================================================================
-- Required DB settings (set externally via ALTER DATABASE postgres SET ...
-- or Supabase Dashboard -> Settings -> Database -> Custom postgres settings):
--   app.supabase_url       — e.g. 'https://vltmrnjsubfzrgrtdqey.supabase.co'
--   app.reconcile_secret   — must match RECONCILE_SECRET env var on edge function
--
-- Operator runbook (run ONCE per environment after this migration):
--   ALTER DATABASE postgres SET app.supabase_url = 'https://<PROJECT_REF>.supabase.co';
--   ALTER DATABASE postgres SET app.reconcile_secret = '<random-32-byte-hex>';
--   SELECT pg_reload_conf();
-- Then deploy fathom-reconcile with RECONCILE_SECRET=<same-value> in its env.

DO $outer$
BEGIN
  PERFORM cron.schedule(
    'fathom-daily-reconcile',
    '0 7 * * *',
    $body$
    SELECT net.http_post(
      url := current_setting('app.supabase_url', true) || '/functions/v1/fathom-reconcile',
      headers := jsonb_build_object(
        'Content-Type', 'application/json',
        'X-Reconcile-Secret', current_setting('app.reconcile_secret', true)
      ),
      body := '{"mode": "reconcile"}'::jsonb
    );
    $body$
  );
  RAISE NOTICE 'Scheduled fathom-daily-reconcile cron (07:00 UTC daily)';
EXCEPTION
  WHEN undefined_function THEN
    RAISE NOTICE 'pg_cron not available - reconcile cron disabled. Use Supabase Scheduled Functions dashboard as fallback.';
  WHEN OTHERS THEN
    RAISE NOTICE 'Failed to schedule fathom-daily-reconcile: %', SQLERRM;
END $outer$;

-- ============================================================================
-- VERIFICATION (run manually after migration + DB setting configuration)
-- ============================================================================
-- 1. Confirm extensions loaded:
--      SELECT extname FROM pg_extension WHERE extname IN ('pg_cron', 'pg_net');
--    Expected: 2 rows.
--
-- 2. Confirm job registered:
--      SELECT jobid, jobname, schedule, active FROM cron.job WHERE jobname = 'fathom-daily-reconcile';
--    Expected: 1 row with schedule '0 7 * * *' and active = true.
--
-- 3. Confirm DB settings:
--      SHOW app.supabase_url;
--      SHOW app.reconcile_secret;
--
-- 4. Manual trigger:
--      SELECT cron.alter_job(
--        job_id := (SELECT jobid FROM cron.job WHERE jobname = 'fathom-daily-reconcile'),
--        schedule := '* * * * *'
--      );
--    Wait 1 minute, check cron.job_run_details for status='succeeded', then revert.
```

```typescript
import { describe, it, expect, beforeAll, afterAll } from 'vitest';
import { createClient, type SupabaseClient } from '@supabase/supabase-js';
import { randomUUID } from 'crypto';

const SUPABASE_URL = process.env.SUPABASE_URL || process.env.VITE_SUPABASE_URL;
const SUPABASE_SERVICE_KEY = process.env.SUPABASE_SERVICE_ROLE_KEY;
const integrationDbReachable = Boolean(SUPABASE_URL && SUPABASE_SERVICE_KEY);

describe.skipIf(!integrationDbReachable)('Phase 39 fathom-daily-reconcile cron + multi-account', () => {
  let supabase: SupabaseClient;
  const testUserId = randomUUID();
  let createdUser = false;
  const sourceIdA = randomUUID();
  const sourceIdB = randomUUID();

  beforeAll(async () => {
    supabase = createClient(SUPABASE_URL!, SUPABASE_SERVICE_KEY!, { auth: { autoRefreshToken: false, persistSession: false } });
    const { error } = await supabase.auth.admin.createUser({
      id: testUserId,
      email: `phase39-cron-${testUserId}@test.local`,
      email_confirm: true,
    } as never);
    if (!error) createdUser = true;
  });

  afterAll(async () => {
    if (!supabase) return;
    await supabase.from('fathom_raw_calls').delete().eq('user_id', testUserId);
    await supabase.from('import_sources').delete().eq('user_id', testUserId);
    if (createdUser) await supabase.auth.admin.deleteUser(testUserId);
  });

  it('multi-account: 2 active import_sources rows persist for a single user', async () => {
    const baseRow = {
      user_id: testUserId,
      source_app: 'fathom',
      is_active: true,
    };
    const { error: errA } = await supabase
      .from('import_sources')
      .insert({ ...baseRow, id: sourceIdA, account_email: 'a@test.local' });
    const { error: errB } = await supabase
      .from('import_sources')
      .insert({ ...baseRow, id: sourceIdB, account_email: 'b@test.local' });
    expect(errA).toBeNull();
    expect(errB).toBeNull();

    const { data: sources, error: selErr } = await supabase
      .from('import_sources')
      .select('id')
      .eq('user_id', testUserId)
      .eq('source_app', 'fathom')
      .eq('is_active', true);
    expect(selErr).toBeNull();
    expect((sources ?? []).length).toBe(2);
  });

  it('reconcile-style query iterates ALL active fathom sources globally', async () => {
    const { data: sources, error } = await supabase
      .from('import_sources')
      .select('id, user_id')
      .eq('source_app', 'fathom')
      .eq('is_active', true);
    expect(error).toBeNull();
    const ids = (sources ?? []).map((s: { id: string }) => s.id);
    expect(ids).toContain(sourceIdA);
    expect(ids).toContain(sourceIdB);
  });

  it('artificial 5-row gap is detectable via mirror upsert (smoke test)', async () => {
    const baseTs = Date.now();
    const seedIds = Array.from({ length: 5 }, (_, i) => baseTs + i);

    const { error: insErr } = await supabase
      .from('fathom_raw_calls')
      .insert(seedIds.map((rid) => ({
        recording_id: rid,
        user_id: testUserId,
        title: `gap-test-${rid}`,
        created_at: new Date().toISOString(),
        import_source_id: sourceIdA,
        mirror_version: 1,
      })));
    expect(insErr).toBeNull();

    const { error: delErr } = await supabase
      .from('fathom_raw_calls')
      .delete()
      .eq('user_id', testUserId)
      .in('recording_id', seedIds);
    expect(delErr).toBeNull();

    const { error: upErr } = await supabase
      .from('fathom_raw_calls')
      .upsert(seedIds.map((rid) => ({
        recording_id: rid,
        user_id: testUserId,
        title: `gap-test-${rid}`,
        created_at: new Date().toISOString(),
        import_source_id: sourceIdA,
        mirror_version: 1,
      })), { onConflict: 'recording_id,user_id' });
    expect(upErr).toBeNull();

    const { count } = await supabase
      .from('fathom_raw_calls')
      .select('*', { count: 'exact', head: true })
      .eq('user_id', testUserId)
      .in('recording_id', seedIds);
    expect(count).toBe(5);
  });
});
```
